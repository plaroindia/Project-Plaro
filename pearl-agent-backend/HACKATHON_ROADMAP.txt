    created_at timestamp with time zone DEFAULT now()

-- 3. Enhance ai_task_results for metrics
ALTER TABLE public.ai_task_results ADD COLUMN (
    module_progress_id uuid REFERENCES ai_module_progress(id),
    time_taken_minutes integer,
    attempt_number integer DEFAULT 1,
    skill_improvement_delta numeric,  -- Confidence change
    is_optimal_solution boolean,
    feedback_quality_score numeric,  -- 0-1 rating
    created_at timestamp with time zone DEFAULT now()

-- 4. Enhance ai_module_progress for better tracking
ALTER TABLE public.ai_module_progress ADD COLUMN (
    practice_task_id uuid REFERENCES ai_practice_tasks(id),
    min_practice_score numeric DEFAULT 60,
    practice_completed boolean DEFAULT false,
    practice_attempts integer DEFAULT 0,
    checkpoint_attempts integer DEFAULT 0,
    skill_confidence_before numeric,
    skill_confidence_after numeric,
    estimated_completion_date date,
    actual_completion_date date,
    completion_velocity numeric  -- days to complete
);

-- 5. Activate ai_roadmap_milestones with module linking
ALTER TABLE public.ai_roadmap_milestones ADD COLUMN (
    module_progress_id uuid REFERENCES ai_module_progress(id),
    completion_percentage numeric DEFAULT 0,
    feasibility_score numeric,  -- 0-1, validates timeline
    estimated_hours integer,
    actual_hours_spent integer DEFAULT 0,
    status text DEFAULT 'pending',
    updated_at timestamp with time zone DEFAULT now()
);

-- 6. Create skill taxonomy table
CREATE TABLE public.skill_taxonomy (
    skill_id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    skill_name text UNIQUE NOT NULL,
    category text NOT NULL,  -- frontend, backend, soft-skills, etc
    proficiency_levels jsonb,  -- beginner, intermediate, advanced
    related_skills text[],
    industry_demand numeric,  -- 0-1
    learning_hours_baseline integer,  -- estimated hours
    created_at timestamp with time zone DEFAULT now()
);

-- 7. Create source tracking table
CREATE TABLE public.learning_sources (
    source_id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id uuid REFERENCES ai_agent_sessions(id),
    source_type text,  -- course, challenge, article, etc
    source_title text NOT NULL,
    source_url text UNIQUE,
    provider text,  -- coursera, udemy, leetcode, etc
    relevance_score numeric,  -- 0-1
    difficulty text,
    estimated_hours integer,
    citations integer DEFAULT 0,
    created_at timestamp with time zone DEFAULT now()
);

-- 8. Create evaluation metrics table
CREATE TABLE public.skill_gap_analysis (
    analysis_id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id uuid REFERENCES ai_agent_sessions(id),
    skill_name text NOT NULL,
    required_proficiency text,  -- from JD
    user_proficiency text,      -- from profile
    gap_level numeric,  -- 0-1, gap severity
    learning_path_feasibility numeric,  -- 0-1
    confidence_score numeric,
    accuracy_score numeric,
    created_at timestamp with time zone DEFAULT now()
);

-- 9. Create reflection & adjustment table
CREATE TABLE public.learning_reflections (
    reflection_id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id uuid REFERENCES ai_agent_sessions(id),
    completed_modules integer,
    avg_checkpoint_score numeric,
    avg_practice_score numeric,
    pace_analysis jsonb,  -- ahead, on-time, behind
    struggling_skills text[],
    mastered_skills text[],
    recommendations jsonb,
    roadmap_adjustments jsonb,
    reflection_timestamp timestamp with time zone DEFAULT now()
);

from enum import Enum

class SkillCategory(str, Enum):
    """Skill taxonomy categories"""
    FRONTEND = "frontend"
    BACKEND = "backend"
    DEVOPS = "devops"
    DATA = "data"
    SOFT_SKILLS = "soft_skills"
    DOMAIN_SPECIFIC = "domain_specific"

class ProficiencyLevel(str, Enum):
    """Skill proficiency levels"""
    BEGINNER = "beginner"
    INTERMEDIATE = "intermediate"
    ADVANCED = "advanced"
    EXPERT = "expert"

class PracticeType(str, Enum):
    """Types of practice tasks"""
    CODING = "coding"
    BEHAVIORAL = "behavioral"
    DECISION = "decision"
    PROJECT = "project"

class SourceType(str, Enum):
    """Learning source types"""
    COURSE = "course"
    CHALLENGE = "challenge"
    ARTICLE = "article"
    DOCUMENTATION = "documentation"
    PROJECT = "project"

class SourceProvider(str, Enum):
    """Authorized providers (no scraping)"""
    COURSERA = "coursera"
    UDEMY = "udemy"
    LEETCODE = "leetcode"
    HACKERRANK = "hackerrank"
    GITHUB = "github"
    MEDIUM = "medium"
    DEV_TO = "dev_to"
    MDN = "mdn"
    OFFICIAL_DOCS = "official_docs"

class FeasibilityStatus(str, Enum):
    """Timeline feasibility"""
    FEASIBLE = "feasible"
    TIGHT = "tight"
    INFEASIBLE = "infeasible"
    NEEDS_ADJUSTMENT = "needs_adjustment"

## 8️⃣ **SPRINT 7: API Integration & Evaluation Framework (Days 7-8)**

### 7.1 Main API Endpoints

File: routes/complete_flow.py

Endpoints:

POST /api/v1/job-to-roadmap
- Input: user_id, jd_text
- Process: Parse JD → Gap analysis → Build roadmap → Create practices
- Output: session_id, roadmap_id, skill_gaps, feasibility_status, first_module
- Validates: No hallucinations, feasible timeline

GET /api/v1/progress/{session_id}
- Output: Overall progress %, modules completed, scores, estimated completion

GET /api/v1/reflection/{session_id}
- Output: Reflection data, recommendations, roadmap adjustments

POST /api/v1/modules/{module_id}/submit-practice
- Input: Solution code/answer + time_taken
- Process: Auto-grade, save result, check completion
- Output: Score, feedback, passed/failed, next action

POST /api/v1/checkpoints/{module_id}/submit
- Input: Quiz answers
- Process: Grade, save result, check if module complete
- Output: Passed/failed, score, feedback

GET /api/v1/dashboard/{session_id}
- Output: All dashboard data (progress, reflection, metrics)

### 7.2 Evaluation Metrics

File: services/evaluation_metrics.py

Metrics:

1. Skill-Gap Accuracy (0-1)
   - User data quality
   - JD extraction quality
   - Prerequisite clarity

2. Roadmap Feasibility (0-1)
   - Timeline validation
   - Hours available vs required
   - Status: feasible/tight/infeasible

3. Practice Diversity (0-1)
   - Type distribution (coding/behavioral/decision)
   - At least 2-3 types represented

4. Measurable Progress (0-1)
   - Checkpoint score improvement
   - Practice score improvement
   - Skills mastered count

Overall Score: Average of all 4 metrics
